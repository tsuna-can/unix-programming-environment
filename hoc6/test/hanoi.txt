proc printRoute(){
  if ($1 == 1) {
    print "from A "
  } else if ($1 == 2){
    print "from B "
  } else if ($1 == 3){
    print "from C "
  }
  if ($2 == 1) {
    print "to A"
  } else if ($2 == 2){
    print "to B"
  } else if ($2 == 3){
    print "to C"
  }
}

proc hanoi() {
  if ($1 > 1) { hanoi($1 - 1, $2, $4, $3) }
    printRoute($2, $4)
    print " : ", $1
  if ($1 > 1) { hanoi($1 - 1, $3, $2, $4) }
}

hanoi(3, 1, 2, 3)
